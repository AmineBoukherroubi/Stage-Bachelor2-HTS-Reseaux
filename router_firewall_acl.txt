enable
configure terminal

! 1. EXCLUSION DHCP (Pour resoudre le conflit d'IP du NVR)
ip dhcp excluded-address 192.168.30.1 192.168.30.10
! Le serveur NVR aura l'adresse fixe 192.168.30.5

ip dhcp pool CCTV_POOL
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
exit

! 2. CREATION DES REGLES DE SECURITE (ACL)
! Autoriser le trafic de l'Administration vers les Cameras
access-list 100 permit ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
! Bloquer tout autre trafic vers les Cameras pour eviter les piratages
access-list 100 deny ip any 192.168.30.0 0.0.0.255
! Autoriser le reste du trafic vers Internet
access-list 100 permit ip any any

! 3. APPLICATION DE L'ACL SUR L'INTERFACE
interface GigabitEthernet0/1.30
 ip access-group 100 in
exit
